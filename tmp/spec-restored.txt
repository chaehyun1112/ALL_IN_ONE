CREATE TABLE TB_EMP (

    EMP_ID VARCHAR(20) PRIMARY KEY,

    EMP_PW VARCHAR(50) NOT NULL,

    HOSP_DIV_ID VARCHAR(30) NOT NULL,

    WARD_ID BIGINT NOT NULL,

    EMP_NM VARCHAR(20) NOT NULL,

    ROLE_CD VARCHAR(10) NOT NULL DEFAULT 'USER',

    AUTH_ST VARCHAR(10) NOT NULL DEFAULT 'PENDING',

    CRT_DT TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UPD_DT TIMESTAMPTZ,

    CONSTRAINT FK_EMP_WARD

        FOREIGN KEY (WARD_ID)

        REFERENCES TB_WARD(WARD_ID),

    CONSTRAINT CK_EMP_ROLE

        CHECK (ROLE_CD IN ('ADMIN', 'USER')),

    CONSTRAINT CK_EMP_AUTH

        CHECK (AUTH_ST IN ('PENDING', 'APPROVED'))

);

CREATE TABLE TB_DEVICE (

    DEV_ID BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    WARD_ID BIGINT NOT NULL,

    DEV_NM VARCHAR(50) NOT NULL,

    INSTALL_LOC VARCHAR(50) NOT NULL,

    SERIAL_NO VARCHAR(50) NOT NULL UNIQUE,

    IP_ADDR INET,

    CONN_ST VARCHAR(10) NOT NULL DEFAULT 'OFFLINE',

    LAST_COMM_DT TIMESTAMPTZ,

    INSTALL_DT TIMESTAMPTZ,

    LAST_CHECK_DT TIMESTAMPTZ,

    CONSTRAINT FK_DEVICE_WARD

        FOREIGN KEY (WARD_ID)

        REFERENCES TB_WARD(WARD_ID),

    CONSTRAINT CK_DEVICE_CONN

        CHECK (CONN_ST IN ('ONLINE', 'OFFLINE'))

);
